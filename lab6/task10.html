Напишіть функцію checkBrackets(str) яка приймає рядок жс коду (someFn) і перевіряє правильність закриття дужок () {} [] Якщо рядок містить коректний код функція повертає true. В іншому випадку повертає false 


        function checkBrackets(str) {
            const stack = [];
            const brackets = {
                ')': '(',
                '}': '{',
                ']': '['
            };
        
            for (const char of str) {
                if (['(', '{', '['].includes(char)) {
                    stack.push(char);
                } else if ([')', '}', ']'].includes(char)) {
                    if (stack.pop() !== brackets[char]) {
                        return false;
                    }
                }
            }
        
            return stack.length === 0;
        }
        
        function checkCodeForBrackets() {
            const codeInput = prompt("Введіть рядок JavaScript-коду для перевірки дужок:");
        
            if (codeInput !== null) {
                const isValid = checkBrackets(codeInput);
        
                console.log(isValid
                    ? "Код містить правильні дужки."
                    : "Код містить помилки в дужках.");
            } else {
                console.log("Операція була скасована користувачем.");
            }
        }
